# ideology/no-repeated-literals

`src/notifications.js` · `sendNotification`, `publishEvent` · a9d7f79 → prism-style simulation

**Rule:** Name this repeated literal once as a constant and reference it.

## Before

```js
  for (let attempt = 0; attempt < 3; attempt++) {
    try {
      const response = await fetch(`${NOTIFICATIONS_URL}/send`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'x-api-key': apiKey
        },
        body: JSON.stringify({ to, subject, message })
      });

      if (response.status === 500) {
        lastError = new Error('Provider returned 500');
        continue;
      }
      // ...
  }

  console.error('Failed to send notification after 3 attempts:', lastError);
```

## After

```js
// Notification service endpoints
const SEND_ENDPOINT = '/send';
const PUBLISH_TOPICS_PATH = '/topics';

// HTTP constants for outbound calls
const POST_METHOD = 'POST';
const CONTENT_TYPE_HEADER = 'Content-Type';
const JSON_CONTENT_TYPE = 'application/json';
const API_KEY_HEADER = 'x-api-key';

// Service response codes
const PROVIDER_ERROR_STATUS = 500;

// Retry policy
const MAX_SEND_ATTEMPTS = 3;

  for (let attempt = 0; attempt < MAX_SEND_ATTEMPTS; attempt++) {
    try {
      const response = await fetch(`${NOTIFICATIONS_URL}${SEND_ENDPOINT}`, {
        method: POST_METHOD,
        headers: {
          [CONTENT_TYPE_HEADER]: JSON_CONTENT_TYPE,
          [API_KEY_HEADER]: apiKey
        },
        body: JSON.stringify({ to, subject, message })
      });

      if (response.status === PROVIDER_ERROR_STATUS) {
        lastError = new Error(`Provider returned ${PROVIDER_ERROR_STATUS}`);
        continue;
      }
      // ...
  }

  console.error(`Failed to send notification after ${MAX_SEND_ATTEMPTS} attempts:`, lastError);
```
