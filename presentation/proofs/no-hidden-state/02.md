# ideology/no-hidden-state

`src/db.js` · `makeRequest`, `getDatabaseName` · a9d7f79 → 7ce6724

**Rule:** Pass this state in as a parameter instead of reading or writing module scope.

## Before

```js
let baseURL;
let databaseName;

if (process.env.DATABASE_URL) {
  baseURL = process.env.DATABASE_URL;
  databaseName = 'custom';
} else {
  const databaseKind = process.env.DATABASE_KIND || 'memory';
  if (databaseKind === 'file') {
    baseURL = 'http://localhost:4003';
    databaseName = 'file';
  } else {
    baseURL = 'http://localhost:4001';
    databaseName = 'memory';
  }
}

async function makeRequest(method, path, body = null) {
  const url = `${baseURL}${path}`;
  // ...
}

function getDatabaseName() {
  return databaseName;
}
```

## After

```js
function createDbModule() {
  let baseURL;
  let databaseName;

  if (process.env.DATABASE_URL) {
    baseURL = process.env.DATABASE_URL;
    databaseName = 'custom';
  } else {
    // ...same branching on DATABASE_KIND
  }

  async function makeRequest(method, path, body = null) {
    const url = `${baseURL}${path}`;
    // ...
  }

  function getDatabaseName() {
    return databaseName;
  }

  // getAll, getById, insert, update, remove: same bodies

  return { getAll, getById, insert, update, remove, getDatabaseName, baseURL };
}

const dbInstance = createDbModule();
```
