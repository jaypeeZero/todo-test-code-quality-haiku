# ideology/no-repeated-literals

`src/index.js` · `/me/current-project` handlers · a9d7f79 → 7ce6724

**Rule:** Name this repeated literal once as a constant and reference it.

## Before

```js
app.put('/me/current-project', requireAuth, async (req, res) => {
  // ...
  const project = await getById('projects', project_id);
  // ...
});

app.get('/me/current-project', requireAuth, async (req, res) => {
  // ...
  const project = await getById('projects', projectId);
  // ...
});
```

## After

```js
const PROJECTS_TABLE = 'projects';
const CURRENT_PROJECT_ENDPOINT = '/me/current-project';

app.put(CURRENT_PROJECT_ENDPOINT, requireAuth, async (req, res) => {
  // ...
  const project = await getById(PROJECTS_TABLE, project_id);
  // ...
});

app.get(CURRENT_PROJECT_ENDPOINT, requireAuth, async (req, res) => {
  // ...
  const project = await getById(PROJECTS_TABLE, projectId);
  // ...
});
```
